/**
 * sendChatPlaceholder - A placeholder function for sending chat messages.
 */
function sendChatPlaceholder() {
  let inputField = document.getElementById("chatInputField");
  let message = inputField.value.trim();
  if (message.length > 0) {
    addChatMessage("You", message);
    inputField.value = "";
  }
}

/**
 * sendMessagePlaceholder - Trigger sending on Enter key press.
 */
function sendMessagePlaceholder(event) {
  if (event.key === "Enter") {
    sendChatPlaceholder();
  }
}

/**
 * addChatMessage - Adds a new message to the chat box.
 */
function addChatMessage(user, message) {
  const chatBox = document.getElementById("chatBox");
  if (!chatBox) return;

  let messageDiv = document.createElement("div");
  messageDiv.classList.add("chat-message", "user-message");
  messageDiv.textContent = `${user}: ${message}`;
  chatBox.appendChild(messageDiv);

  // Auto-scroll to bottom
  chatBox.scrollTop = chatBox.scrollHeight;
}
